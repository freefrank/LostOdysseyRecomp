cmake_minimum_required(VERSION 3.24)
project(LoSrHybridCpu LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
enable_testing()
add_executable(sr_hybrid_motion_test "${ROOT}/tools/tests/sr_hybrid_motion_test.cpp"
    "${ROOT}/LostOdysseyRecomp/gpu/upscaling_plan.cpp")
target_include_directories(sr_hybrid_motion_test PRIVATE "${ROOT}/LostOdysseyRecomp")
if(MSVC)
  target_compile_options(sr_hybrid_motion_test PRIVATE /utf-8 /W4)
else()
  target_compile_options(sr_hybrid_motion_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
option(LO_HYBRID_SANITIZE "Enable address/undefined sanitizers for CPU tests" OFF)
if(LO_HYBRID_SANITIZE AND NOT MSVC)
  target_compile_options(sr_hybrid_motion_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(sr_hybrid_motion_test PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME sr_hybrid_motion_cpu COMMAND sr_hybrid_motion_test)
