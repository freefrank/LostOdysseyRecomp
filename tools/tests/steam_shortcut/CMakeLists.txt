cmake_minimum_required(VERSION 3.20)
project(LoSteamShortcutTests LANGUAGES CXX)
enable_testing()
get_filename_component(LO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(LoSteamShortcutTest "${LO_ROOT}/tools/tests/steam_shortcut_test.cpp"
    "${LO_ROOT}/tools/tests/steam_shortcut_artwork_stub.cpp"
    "${LO_ROOT}/LostOdysseyRecomp/os/steam_shortcut.cpp")
target_include_directories(LoSteamShortcutTest PRIVATE "${LO_ROOT}/LostOdysseyRecomp")
target_compile_features(LoSteamShortcutTest PRIVATE cxx_std_20)
if(WIN32)
    target_link_libraries(LoSteamShortcutTest PRIVATE advapi32)
endif()
if(MSVC)
    target_compile_options(LoSteamShortcutTest PRIVATE /utf-8 /UNDEBUG)
else()
    target_compile_options(LoSteamShortcutTest PRIVATE -Wall -Wextra -Wpedantic -UNDEBUG)
endif()
add_test(NAME steam_shortcut COMMAND LoSteamShortcutTest)
set_tests_properties(steam_shortcut PROPERTIES TIMEOUT 30)
