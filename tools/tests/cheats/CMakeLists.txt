cmake_minimum_required(VERSION 3.20)
project(LoCheatTests LANGUAGES CXX)
# Standalone: no game images, generated PPC, GPU SDKs or SDL installation.
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
find_package(Threads REQUIRED)
include(CTest)
option(LO_CHEATS_SANITIZE "Enable ASan/UBSan on Clang/GCC" OFF)
foreach(kind IN ITEMS core ui)
    add_executable(LoCheats_${kind} ${kind}_test.cpp)
    target_include_directories(LoCheats_${kind} PRIVATE "${REPO_ROOT}/LostOdysseyRecomp")
    target_compile_features(LoCheats_${kind} PRIVATE cxx_std_20)
    target_link_libraries(LoCheats_${kind} PRIVATE Threads::Threads)
    target_compile_definitions(LoCheats_${kind} PRIVATE NOMINMAX)
    if(MSVC)
        target_compile_options(LoCheats_${kind} PRIVATE /utf-8 /W4 /UNDEBUG)
    else()
        target_compile_options(LoCheats_${kind} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        if(LO_CHEATS_SANITIZE)
            target_compile_options(LoCheats_${kind} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(LoCheats_${kind} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME cheats_core COMMAND LoCheats_core)
add_test(NAME cheats_ui COMMAND LoCheats_ui "${CMAKE_CURRENT_BINARY_DIR}/ui")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(SANITIZER_ARG)
    if(LO_CHEATS_SANITIZE)
        set(SANITIZER_ARG --sanitize)
    endif()
    add_test(NAME cheats_runtime_boundary COMMAND "${Python3_EXECUTABLE}"
        "${CMAKE_CURRENT_LIST_DIR}/runtime_test.py" --root "${REPO_ROOT}"
        --cxx "${CMAKE_CXX_COMPILER}" --out "${CMAKE_CURRENT_BINARY_DIR}/adapter" ${SANITIZER_ARG})
endif()
set_tests_properties(cheats_core cheats_ui PROPERTIES TIMEOUT 30)
