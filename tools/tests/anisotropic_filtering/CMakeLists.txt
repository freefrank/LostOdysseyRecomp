cmake_minimum_required(VERSION 3.20)
project(LoAnisotropicFilteringTests LANGUAGES C CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
include(CTest)
option(LO_AF_SANITIZE "Address/undefined behavior sanitizers" OFF)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
set(FIXTURE "${CMAKE_CURRENT_BINARY_DIR}/renderer_fixture.cpp")
add_custom_command(OUTPUT "${FIXTURE}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/generate_renderer_fixture.py"
        --root "${ROOT}" --output "${FIXTURE}"
    DEPENDS generate_renderer_fixture.py "${ROOT}/LostOdysseyRecomp/gpu/renderer.cpp"
    VERBATIM)
add_executable(LoSamplerPaletteTest palette_test.cpp)
add_executable(LoSamplerRendererTest "${FIXTURE}")
foreach(target IN ITEMS LoSamplerPaletteTest LoSamplerRendererTest)
    target_include_directories(${target} PRIVATE "${ROOT}/LostOdysseyRecomp" "${ROOT}/thirdparty/plume")
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE NOMINMAX)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
        if(LO_AF_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endforeach()
add_test(NAME sampler_palette COMMAND LoSamplerPaletteTest)
add_test(NAME sampler_renderer COMMAND LoSamplerRendererTest)
option(LO_AF_VULKAN_TEST "Build real Vulkan dispatch/readback fixture" OFF)
if(LO_AF_VULKAN_TEST)
    if(NOT UNIX OR APPLE)
        message(FATAL_ERROR "This Vulkan fixture lane currently targets Linux")
    endif()
    find_program(GLSLANG glslangValidator REQUIRED)
    find_package(X11 REQUIRED)
    add_subdirectory("${ROOT}/thirdparty/plume" plume)
    target_compile_definitions(plume PRIVATE VULKAN_VALIDATION_LAYER_ENABLED=1)
    add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/sample.spv"
        COMMAND "${GLSLANG}" -V --target-env vulkan1.2 -o "${CMAKE_CURRENT_BINARY_DIR}/sample.spv" "${CMAKE_CURRENT_LIST_DIR}/sample.comp"
        DEPENDS sample.comp VERBATIM)
    add_executable(LoSamplerVulkanTest vulkan_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/sample.spv")
    target_compile_features(LoSamplerVulkanTest PRIVATE cxx_std_20)
    target_include_directories(LoSamplerVulkanTest PRIVATE "${ROOT}/LostOdysseyRecomp" ${X11_INCLUDE_DIR})
    target_link_libraries(LoSamplerVulkanTest PRIVATE plume ${X11_LIBRARIES} ${CMAKE_DL_LIBS})
    add_test(NAME sampler_vulkan COMMAND LoSamplerVulkanTest "${CMAKE_CURRENT_BINARY_DIR}/sample.spv")
    set_tests_properties(sampler_vulkan PROPERTIES TIMEOUT 90)
endif()
