cmake_minimum_required(VERSION 3.20)
project(LoFastForwardAudioTests LANGUAGES CXX)

enable_testing()
add_executable(LoFastForwardAudioTest ../fast_forward_audio_test.cpp)
target_compile_features(LoFastForwardAudioTest PRIVATE cxx_std_20)
target_include_directories(LoFastForwardAudioTest PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../../../LostOdysseyRecomp")
set_target_properties(LoFastForwardAudioTest PROPERTIES CXX_SCAN_FOR_MODULES OFF)
if(MSVC)
    target_compile_options(LoFastForwardAudioTest PRIVATE /UNDEBUG /utf-8)
else()
    target_compile_options(LoFastForwardAudioTest PRIVATE -UNDEBUG)
endif()
add_test(NAME fast_forward_audio COMMAND LoFastForwardAudioTest)
set_tests_properties(fast_forward_audio PROPERTIES TIMEOUT 120)
