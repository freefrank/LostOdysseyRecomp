cmake_minimum_required(VERSION 3.20)
project(LoRectListTests LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
add_executable(LoRectListTest "${ROOT}/tools/tests/rect_list_test.cpp")
target_include_directories(LoRectListTest PRIVATE "${ROOT}/LostOdysseyRecomp")
add_test(NAME rect_list_indices COMMAND LoRectListTest)
option(LO_RECT_LIST_GPU_TESTS "Build real translated-shader/Vulkan readback tests" OFF)
if(LO_RECT_LIST_GPU_TESTS)
    find_package(Vulkan REQUIRED)
    add_subdirectory("${ROOT}/tools/XenosRecomp/thirdparty/fmt" fmt EXCLUDE_FROM_ALL)
    add_library(rect_shaders STATIC
        "${ROOT}/LostOdysseyRecomp/gpu/shader/xenos_translator.cpp"
        "${ROOT}/LostOdysseyRecomp/gpu/shader/dxc_compiler.cpp")
    target_include_directories(rect_shaders PUBLIC "${ROOT}/LostOdysseyRecomp"
        "${ROOT}/tools/XenosRecomp/thirdparty/dxc-bin/inc")
    target_compile_definitions(rect_shaders PUBLIC NOMINMAX)
    target_link_libraries(rect_shaders PUBLIC fmt::fmt ${CMAKE_DL_LIBS})
    add_executable(LoRectListGpuTest "${ROOT}/tools/tests/rect_list_gpu_test.cpp")
    target_link_libraries(LoRectListGpuTest PRIVATE rect_shaders Vulkan::Vulkan)
    add_test(NAME rect_list_gpu COMMAND LoRectListGpuTest)
    set_tests_properties(rect_list_gpu PROPERTIES TIMEOUT 120)
endif()
