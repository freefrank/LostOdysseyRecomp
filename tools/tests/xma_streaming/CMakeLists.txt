cmake_minimum_required(VERSION 3.20)
project(LoXmaStreaming LANGUAGES CXX)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(LoXmaStreamingTest fixture.cpp)
target_compile_features(LoXmaStreamingTest PRIVATE cxx_std_20)
target_include_directories(LoXmaStreamingTest BEFORE PRIVATE stubs "${ROOT}/LostOdysseyRecomp")
if(NOT MSVC)
    # Count production ProcessPacket entries without altering its body.
    target_compile_options(LoXmaStreamingTest PRIVATE -finstrument-functions)
    target_compile_definitions(LoXmaStreamingTest PRIVATE LO_COUNT_CALLS=1)
endif()
option(LO_XMA_SANITIZE "Enable address and undefined sanitizers" ${LO_REVIEW_SANITIZE})
if(LO_XMA_SANITIZE AND NOT MSVC)
    target_compile_options(LoXmaStreamingTest PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(LoXmaStreamingTest PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME xma_streaming COMMAND LoXmaStreamingTest)
