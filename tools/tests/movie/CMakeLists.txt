cmake_minimum_required(VERSION 3.20)
project(LoMovieContracts LANGUAGES CXX)
enable_testing()
get_filename_component(LO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(LO_RUNTIME "${LO_ROOT}/LostOdysseyRecomp")
option(LO_MOVIE_SANITIZE "Run data/layout checks with ASan and UBSan" ON)
set(LO_MOVIE_JNI_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(LO_MOVIE_JNI_DEFAULT ON)
endif()
option(LO_MOVIE_JNI_TESTS "Run the scripted JNI protocol checks (Linux JDK)" ${LO_MOVIE_JNI_DEFAULT})
add_executable(LoMovieDataTest ../movie_data_test.cpp
    "${LO_RUNTIME}/movie/byte_source.cpp" "${LO_RUNTIME}/movie/mp4_demux.cpp" "${LO_RUNTIME}/modding/mod_api.cpp")
target_include_directories(LoMovieDataTest PRIVATE "${LO_RUNTIME}")
target_compile_features(LoMovieDataTest PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(LoMovieDataTest PRIVATE /UNDEBUG)
else()
    target_compile_options(LoMovieDataTest PRIVATE -UNDEBUG)
    if(LO_MOVIE_SANITIZE)
        target_compile_options(LoMovieDataTest PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(LoMovieDataTest PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME LoMovieDataTest COMMAND LoMovieDataTest
    "${LO_ROOT}/tools/tests/data/testsrc-64x64-bframes.mp4"
    "${LO_ROOT}/tools/tests/data/testsrc-64x64-bframes-frag.mp4")
find_package(Threads REQUIRED)
add_executable(LoMovieSessionTest ../movie_session_test.cpp
    "${LO_RUNTIME}/movie/byte_source.cpp" "${LO_RUNTIME}/movie/mp4_demux.cpp" "${LO_RUNTIME}/modding/mod_api.cpp")
target_include_directories(LoMovieSessionTest PRIVATE "${LO_RUNTIME}" "${LO_ROOT}/tools/XenonRecomp/thirdparty/fmt/include")
target_compile_features(LoMovieSessionTest PRIVATE cxx_std_20)
target_compile_definitions(LoMovieSessionTest PRIVATE FMT_HEADER_ONLY)
target_link_libraries(LoMovieSessionTest PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(LoMovieSessionTest PRIVATE /UNDEBUG)
else()
    target_compile_options(LoMovieSessionTest PRIVATE -UNDEBUG)
    if(LO_MOVIE_SANITIZE)
        target_compile_options(LoMovieSessionTest PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(LoMovieSessionTest PRIVATE -fsanitize=address,undefined)
    endif()
endif()
add_test(NAME LoMovieSessionTest COMMAND LoMovieSessionTest "${LO_ROOT}/tools/tests/data/testsrc-64x64-bframes.mp4")
set_tests_properties(LoMovieSessionTest PROPERTIES TIMEOUT 10)
if(LO_MOVIE_JNI_TESTS)
    find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
    find_package(Java 17 REQUIRED COMPONENTS Runtime)
    find_package(JNI REQUIRED COMPONENTS JVM)
    set(jni_args)
    foreach(path IN LISTS JNI_INCLUDE_DIRS)
        list(APPEND jni_args --jni-include "${path}")
    endforeach()
    add_test(NAME LoMovieJniProtocol COMMAND "${Python3_EXECUTABLE}" -B
        "${LO_ROOT}/tools/tests/movie_jni_test.py" --cxx "${CMAKE_CXX_COMPILER}" ${jni_args})
    set_tests_properties(LoMovieJniProtocol PROPERTIES TIMEOUT 60)
endif()
